\section{Tema d'esame del 8 Settembre 2025}
\subsection*{Esercizio 1}

Si indichi, per ciascuna delle seguenti formule, se sono valide, insoddisfacibili o solo soddisfacibili. In ciascun caso, motivare semanticamente la risposta data. Esibire, successivamente, una deduzione in Natural Deduction per ogni formula che risulta valida o per la sua negazione, se la formula risulta insoddisfacibile.

\paragraph{Svolgimento:}

\subsubsection*{(a) \((P\vee Q)\vee\neg(P\vee Q)\)}
\paragraph{Svolgimento:}
La formula è valida perché equivalente alla forma tautologica \((A\vee\neg A)\) applicando la sostituzione \(\sigma=\{A\mapsto P\vee Q\}\).

\begin{prooftree}
  \AxiomC{\({[\neg ((P\vee Q)\vee\neg(P\vee Q))]}_1\)}
  \AxiomC{\({[P\vee Q]}_2\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\((P\vee Q)\vee\neg(P\vee Q)\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\neg I_2\)}
  \UnaryInfC{\(\neg(P\vee Q)\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\((P\vee Q)\vee\neg(P\vee Q)\)}
  \AxiomC{\({[\neg ((P\vee Q)\vee\neg(P\vee Q))]}_1\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\RAA_1\)}
  \UnaryInfC{\((P\vee Q)\vee\neg(P\vee Q)\)}
\end{prooftree}

\subsubsection*{(b) \((P\wedge\neg Q)\rightarrow\neg(\neg P\vee Q)\)}
\paragraph{Svolgimento:}
La formula è valida, in quanto equivalente, per de Morgan, a \((P\wedge\neg Q)\rightarrow(P\wedge\neg Q)\).

\begin{prooftree}
  \small
  \AxiomC{\({[\neg P\vee Q]}_3\)}
  \AxiomC{\({[\neg P]}_2\)}
  \AxiomC{\({[P\wedge\neg Q]}_1\)}
  \RuleLabel{\(\land E\)}
  \UnaryInfC{\(P\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  
  \AxiomC{\({[Q]}_2\)}
  \AxiomC{\({[P\wedge\neg Q]}_1\)}
  \RuleLabel{\(\land E\)}
  \UnaryInfC{\(\neg Q\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  
  \RuleLabel{\(\lor E_2\)}
  \TrinaryInfC{\(\bot\)}
  \RuleLabel{\(\neg I_3\)}
  \UnaryInfC{\(\neg(\neg P\vee Q)\)}
  \RuleLabel{\(\to I_1\)}
  \UnaryInfC{\((P\wedge\neg Q)\rightarrow\neg(\neg P\vee Q)\)}
\end{prooftree}

\subsubsection*{(c) \((P\vee Q)\rightarrow\neg(\neg P\vee\neg Q)\)}
\paragraph{Svolgimento:}
La formula è solo soddisfacibile, ma non valida. Si osserva che è equivalente a \((P\vee Q)\rightarrow(P\wedge Q)\). È vera se \(P = Q\), ma è falsa per \(P \neq Q\).

\subsubsection*{(d) \(\neg(\neg P\vee\neg Q)\rightarrow(P\vee Q)\)}
\paragraph{Svolgimento:}
La formula è valida: l'antecedente è equivalente a \((P\wedge Q)\), il che implica banalmente il conseguente \((P\vee Q)\). È falsa se e solo se \(P=0\) e \(Q=0\), ma in tal caso l'antecedente è già falso.

\begin{prooftree}
  \AxiomC{\({[\neg(\neg P\vee\neg Q)]}_1\)}
  \AxiomC{\({[\neg P]}_2\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\(\neg P\vee\neg Q\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\RAA_2\)}
  \UnaryInfC{\(P\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\(P\vee Q\)}
  \RuleLabel{\(\to I_1\)}
  \UnaryInfC{\(\neg(\neg P\vee\neg Q)\rightarrow(P\vee Q)\)}
\end{prooftree}

\subsubsection*{(e) \((\neg P\vee\neg Q)\vee(P\wedge Q)\)}
\paragraph{Svolgimento:}
La formula è valida, poiché è semanticamente equivalente a \(\neg(P\wedge Q)\vee(P\wedge Q)\), ossia un'istanza del terzo escluso.

\begin{prooftree}
  \tiny
  \AxiomC{\({[\neg((\neg P\vee\neg Q)\vee(P\wedge Q))]}_1\)}
  \AxiomC{\({[\neg P]}_2\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\(\neg P\vee\neg Q\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\((\neg P\vee\neg Q)\vee(P\wedge Q)\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\RAA_2\)}
  \UnaryInfC{\(P\)}

  \AxiomC{\({[\neg((\neg P\vee\neg Q)\vee(P\wedge Q))]}_1\)}
  \AxiomC{\({[\neg Q]}_3\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\(\neg P\vee\neg Q\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\((\neg P\vee\neg Q)\vee(P\wedge Q)\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\RAA_3\)}
  \UnaryInfC{\(Q\)}

  \RuleLabel{\(\land I\)}
  \BinaryInfC{\(P\wedge Q\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\((\neg P\vee\neg Q)\vee(P\wedge Q)\)}
  
  \AxiomC{\({[\neg((\neg P\vee\neg Q)\vee(P\wedge Q))]}_1\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\RAA_1\)}
  \UnaryInfC{\((\neg P\vee\neg Q)\vee(P\wedge Q)\)}
\end{prooftree}

\subsection*{Esercizio 2}


Si traducano in logica del prim'ordine le seguenti affermazioni in linguaggio naturale, utilizzando solo i predicati indicati di seguito: \(P(\cdot) =\) \q{è una persona}; \(E(\cdot) =\) \q{è un esame}; \(Sc(\cdot) =\) \q{è una disciplina scientifica}; \(Um(\cdot) =\) \q{è una disciplina umanistica}; \(S(\cdot, \cdot) =\) \q{sostiene}; \(St(\cdot, \cdot) =\) \q{studia}; \(M(\cdot) =\) \q{è MATEMATICA}; \(L(\cdot) =\) \q{è LETTERE}.
\paragraph{Svolgimento:}
\begin{itemize}
  \item \textbf{(a)} Le persone che studiano discipline scientifiche sostengono tutti lo stesso esame di matematica.
  \[\exists e\st (E(e) \wedge M(e) \wedge \forall p((P(p) \wedge \exists d\st (Sc(d) \wedge St(p,d))) \rightarrow S(p,e)))\]

  \item \textbf{(b)} Le persone che studiano discipline umanistiche sostengono almeno un esame di lettere.
  \[\forall p((P(p) \wedge \exists d\st (Um(d) \wedge St(p,d))) \rightarrow \exists e\st (E(e) \wedge L(e) \wedge S(p,e)))\]

  \item \textbf{(c)} Le discipline scientifiche vengono studiate da alcune persone che sostengono almeno tre esami di matematica.
  \[\forall d(Sc(d) \rightarrow \exists p\st (P(p) \wedge St(p,d) \wedge \exists e_1, e_2, e_3\st (E(e_1) \wedge E(e_2) \wedge E(e_3) \wedge M(e_1) \wedge M(e_2) \wedge M(e_3) \wedge S(p,e_1) \wedge S(p,e_2) \wedge S(p,e_3) \wedge \neg(e_1=e_2) \wedge \neg(e_2=e_3) \wedge \neg(e_3=e_1))))\]

  \item \textbf{(d)} Le persone che studiano discipline scientifiche sostengono gli stessi esami di matematica. (Soluzione 1)
  \[\forall p_1 \forall p_2 ((P(p_1) \wedge P(p_2) \wedge \exists d\st (Sc(d) \wedge St(p_1,d)) \wedge \exists d\st (Sc(d) \wedge St(p_2,d))) \rightarrow \forall e\st ((E(e) \wedge M(e)) \rightarrow (S(p_1,e) \leftrightarrow S(p_2,e))))\]

  \item \textbf{(e)} Le persone che studiano discipline scientifiche sostengono al massimo due esami di matematica.
  \[\forall p((P(p) \wedge \exists d\st (Sc(d) \wedge St(p,d))) \rightarrow \forall e_1 \forall e_2 \forall e_3 ((E(e_1) \wedge E(e_2) \wedge E(e_3) \wedge S(p,e_1) \wedge S(p,e_2) \wedge S(p,e_3) \wedge M(e_1) \wedge M(e_2) \wedge M(e_3)) \rightarrow (e_1=e_2 \vee e_2=e_3 \vee e_1=e_3)))\]
\end{itemize}

\subsection*{Esercizio 3}


Si indichi, per ciascuna delle seguenti formule se sono valide, insoddisfacibili o solo soddisfacibili. In ciascun caso, motivare la risposta data. Esibire, successivamente, una deduzione in Natural Deduction per ogni formula che risulta valida o per la sua negazione, se la formula risulta insoddisfacibile.


\subsubsection*{(a) \(\forall x\st P(x)\rightarrow\neg\neg\exists x\st \neg P(x)\)}
\paragraph{Svolgimento:}
La formula è equivalente a \(\forall x\st P(x) \rightarrow \exists x\st \neg P(x)\) e conseguentemente a \(\neg\forall x\st P(x) \vee \exists x\st \neg P(x)\), che collassa a \(\neg\forall x\st P(x)\). È dunque una formula soddisfacibile ma non valida.

\subsubsection*{(b) \(\forall x\st R(x,x)\vee\neg\forall x\st \forall y\st R(x,y)\)}
\paragraph{Svolgimento:}
La formula è valida. Si motiva osservando che se la relazione è vera per ogni coppia \((x,y)\), allora è necessariamente vera anche per le coppie riflessive \((x,x)\).

\begin{prooftree}
  \small
  \AxiomC{\({[\neg(\forall x\st R(x,x)\vee\neg\forall x\st \forall y\st R(x,y))]}_1\)}
  \AxiomC{\({[\forall x\st \forall y\st R(x,y)]}_2\)}
  \RuleLabel{\(\forall E\)}
  \UnaryInfC{\(\forall y\st R(z,y)\)}
  \RuleLabel{\(\forall E\)}
  \UnaryInfC{\(R(z,z)\)}
  \RuleLabel{\(\forall I\)}
  \UnaryInfC{\(\forall x\st R(x,x)\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\(\forall x\st R(x,x)\vee\neg\forall x\st \forall y\st R(x,y)\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\neg I_2\)}
  \UnaryInfC{\(\neg\forall x\st \forall y\st R(x,y)\)}
  \RuleLabel{\(\lor I\)}
  \UnaryInfC{\(\forall x\st R(x,x)\vee\neg\forall x\st \forall y\st R(x,y)\)}
  \AxiomC{\({[\neg(\forall x\st R(x,x)\vee\neg\forall x\st \forall y\st R(x,y))]}_1\)}
  \RuleLabel{\(\neg E\)}
  \BinaryInfC{\(\bot\)}
  \RuleLabel{\(\RAA_1\)}
  \UnaryInfC{\(\forall x\st R(x,x)\vee\neg\forall x\st \forall y\st R(x,y)\)}
\end{prooftree}

\subsubsection*{(c) \(\forall x\st (\exists y\st C(x,y)\rightarrow A(x))\rightarrow\forall x\st \exists y\st (C(x,y)\rightarrow A(x))\)}
\paragraph{Svolgimento:}
La formula è valida: \(\forall x\st (\neg\exists y\st C(x,y) \vee A(x))\) equivale a \(\forall x\st \forall y\st (\neg C(x,y) \vee A(x))\) e quindi \(\forall x\st \forall y\st (C(x,y) \rightarrow A(x))\). Se vale per tutte le coppie, allora vale che per ogni \(x\) trovo un \(y\) che la soddisfa.

\begin{prooftree}
  \AxiomC{\({[\forall x\st (\exists y\st C(x,y)\rightarrow A(x))]}_1\)}
  \RuleLabel{\(\forall E\)}
  \UnaryInfC{\(\exists y\st C(z,y)\rightarrow A(z)\)}
  \AxiomC{\({[C(z,t)]}_2\)}
  \RuleLabel{\(\exists I\)}
  \UnaryInfC{\(\exists y\st C(z,y)\)}
  \RuleLabel{\(\to E\)}
  \BinaryInfC{\(A(z)\)}
  \RuleLabel{\(\to I_2\)}
  \UnaryInfC{\(C(z,t)\rightarrow A(z)\)}
  \RuleLabel{\(\exists I\)}
  \UnaryInfC{\(\exists y\st (C(z,y)\rightarrow A(z))\)}
  \RuleLabel{\(\forall I\)}
  \UnaryInfC{\(\forall x\st \exists y\st (C(x,y)\rightarrow A(x))\)}
  \RuleLabel{\(\to I_1\)}
  \UnaryInfC{\(\forall x\st (\exists y\st C(x,y)\rightarrow A(x))\rightarrow\forall x\st \exists y\st (C(x,y)\rightarrow A(x))\)}
\end{prooftree}

\subsubsection*{(d) \(\exists x\st \forall y\st ((R(x,y)\rightarrow R(y,x))\wedge(R(y,x)\rightarrow R(x,y)))\)}
\paragraph{Svolgimento:}
La formula è soddisfacibile. Si può avere un modello \(M\) dove \(D=\{1\}\) e \(R^M=\{(1,1)\}\) che la soddisfa, mentre un modello \(M'\) con \(D=\{1,2\}\) e \(R^{M'}=\{(1,2)\}\) non la soddisfa. 

\subsubsection*{(e) \((\exists y\st P(c,y)\wedge(d=c))\rightarrow\exists y\st P(d,y)\)}
\paragraph{Svolgimento:}
La formula è valida perché, per la simmetria dell'uguaglianza, \((d=c)\) equivale a \((c=d)\), per cui possiamo operare una sostituzione. 

\begin{prooftree}
  \AxiomC{\({[\exists y\st P(c,y) \wedge (d=c)]}_1\)}
  \RuleLabel{\(\land E\)}
  \UnaryInfC{\(d=c\)}
  
  \AxiomC{\(d=d\)}
  \RuleLabel{\(Refl\)}
  \UnaryInfC{\(d=d \to c=d\)}
  
  \RuleLabel{\(Sub_\phi\)}
  \BinaryInfC{\(c=d\)}
  
  \AxiomC{\({[\exists y\st P(c,y) \wedge (d=c)]}_1\)}
  \RuleLabel{\(\land E\)}
  \UnaryInfC{\(\exists y\st P(c,y)\)}
  
  \RuleLabel{\(Sub_\phi\)}
  \BinaryInfC{\(\exists y\st P(c,y) \to \exists y\st P(d,y)\)}

  \AxiomC{\({[\exists y\st P(c,y) \wedge (d=c)]}_1\)}
  \RuleLabel{\(\land E\)}
  \UnaryInfC{\(\exists y\st P(c,y)\)}

  \RuleLabel{\(\to E\)}
  \BinaryInfC{\(\exists y\st P(d,y)\)}
  \RuleLabel{\(\to I_1\)}
  \UnaryInfC{\((\exists y\st P(c,y)\wedge d=c)\rightarrow\exists y\st P(d,y)\)}
\end{prooftree}
