\section{Tema d'esame del 11 Luglio 2025}
\subsection*{Esercizio 1}
Indicare, per ciascuna delle seguenti formule, se sono valide, insoddisfacibili o solo soddisfacibili, motivando semanticamente la risposta. Esibire una deduzione in Natural Deduction per ogni formula che risulta valida o per la sua negazione, se insoddisfacibile.

\subsubsection*{(a) \((p \land (\neg p \lor q)) \to \neg(p \land \neg q)\)}
\paragraph{Svolgimento:}
La formula è \textbf{valida}. Semanticamente, l'antecedente implica il conseguente poiché \((p \land (\neg p \lor q)) \equiv p \land q\), e questo contraddice \((p \land \neg q)\).

\begin{prooftree}
    \small
    \AxiomC{\({[p \land (\neg p \lor q)]}_1\)}
    \RuleLabel{\(\land E\)}
    \UnaryInfC{\(\neg p \lor q\)}
    \AxiomC{\({[p \land (\neg p \lor q)]}_1\)}
    \RuleLabel{\(\land E\)}
    \UnaryInfC{\(p\)}
    \AxiomC{\({[\neg p]}_2\)}
    \RuleLabel{\(\neg E\)}
    \BinaryInfC{\(\bot\)}
    \AxiomC{\({[p \land \neg q]}_3\)}
    \RuleLabel{\(\land E\)}
    \UnaryInfC{\(\neg q\)}
    \AxiomC{\({[q]}_2\)}
    \RuleLabel{\(\neg E\)}
    \BinaryInfC{\(\bot\)}
    \RuleLabel{\(\lor E_{2}\)}
    \TrinaryInfC{\(\bot\)}
    \RuleLabel{\(\neg I_3\)}
    \UnaryInfC{\(\neg(p \land \neg q)\)}
    \RuleLabel{\(\to I_1\)}
    \UnaryInfC{\((p \land (\neg p \lor q)) \to \neg(p \land \neg q)\)}
\end{prooftree}

\subsubsection*{(b) \(((p \lor r) \lor (p \land q)) \to (\neg r \lor p)\)}
\paragraph{Svolgimento:}
La formula è \textbf{soddisfacibile ma non valida}. La formula è falsa per \(r=1\) e \(p=0\), ma è vera per \(r=0\) o \(p=1\).

\subsection*{Esercizio 2}
Tradurre in logica del prim'ordine con uguaglianza le seguenti affermazioni in linguaggio naturale, utilizzando solo i simboli di relazione aggiuntivi indicati di seguito: \(C(\cdot) = \) \q{\(\cdot\) è un cane}; \(P(\cdot) = \) \q{\(\cdot\) è una persona}; \(PR(\cdot,\circ) = \) \q{\(\cdot\) è proprietario di \(\circ\)}; \(F(\cdot,\circ) = \) \q{\(\cdot\) è fedele a \(\circ\)}; \(A(\cdot,\circ) = \) \q{\(\cdot\) ama \(\circ\)}.
\paragraph{Svolgimento:}
\begin{itemize}
    \item \textbf{(a)} I cani sono fedeli ai loro proprietari.
    \[\forall c (C(c) \to \forall p ((P(p) \land PR(p,c)) \to F(c,p)))\]

    \item \textbf{(b)} Ci sono cani fedeli a uno dei due proprietari.
    \[\exists c (C(c) \land \exists p (P(p) \land PR(p,c) \land F(c,p) \land \forall q ((P(q) \land PR(q,c) \land F(c,q)) \to q = p)))\]

    \item \textbf{(c)} Nessun cane è infedele a tutti i suoi proprietari.
    \[\neg \exists c (C(c) \land \forall p ((P(p) \land PR(p,c)) \to \neg F(c,p)))\]

    \item \textbf{(d)} Tutti quelli che amano gli animali possiedono due cani.
    \[\forall p ((P(p) \land \forall c (C(c) \to A(p,c))) \to \exists c_1 \exists c_2 (C(c_1) \land C(c_2) \land PR(p,c_1) \land PR(p,c_2) \land \neg(c_1 = c_2)))\]

    \item \textbf{(e)} Non ci sono più di due cani posseduti dallo stesso proprietario che sono fedeli ad altre persone.
    \[\forall p (P(p) \to \forall c_1 \forall c_2 \forall c_3 ((C(c_1) \land C(c_2) \land C(c_3) \land PR(p,c_1) \land PR(p,c_2) \land PR(p,c_3) \land \exists q_1 (P(q_1) \land F(c_1,q_1) \land \neg(p=q_1)) \land \exists q_2 (P(q_2) \land F(c_2,q_2) \land \neg(p=q_2)) \land \exists q_3 (P(q_3) \land F(c_3,q_3) \land \neg(p=q_3))) \to (c_1=c_2 \lor c_2=c_3 \lor c_1=c_3)))\]
\end{itemize}

\subsection*{Esercizio 3}

\subsubsection*{(a) \(\exists x \st \neg P(x) \land \neg\neg\forall x \st P(x)\)}
\paragraph{Svolgimento:}
\textbf{Insoddisfacibile}. Dimostriamo la sua negazione.
\begin{prooftree}
    \small
    \AxiomC{\({[\exists x \st \neg P(x) \land \neg\neg\forall x \st P(x)]}_1\)}
    \RuleLabel{\(\land E\)}
    \UnaryInfC{\(\exists x \st \neg P(x)\)}
    \AxiomC{\({[\exists x \st \neg P(x) \land \neg\neg\forall x \st P(x)]}_1\)}
    \RuleLabel{\(\land E\)}
    \UnaryInfC{\(\neg\neg\forall x \st P(x)\)}
    \RuleLabel{\(\RAA\)}
    \UnaryInfC{\(\forall x \st P(x)\)}
    \RuleLabel{\(\forall E\)}
    \UnaryInfC{\(P(z)\)}
    \AxiomC{\({[\neg P(z)]}_2\)}
    \RuleLabel{\(\neg E\)}
    \BinaryInfC{\(\bot\)}
    \RuleLabel{\(\exists E_2\)}
    \BinaryInfC{\(\bot\)}
    \RuleLabel{\(\neg I_1\)}
    \UnaryInfC{\(\neg(\exists x \st \neg P(x) \land \neg\neg\forall x \st P(x))\)}
\end{prooftree}

\subsubsection*{(b) \((\exists y \st (\forall x \st P(x,y) \to B(y))) \to (\exists y \st (\exists x \st P(x,y) \to B(y)))\)}
\paragraph{Svolgimento:}
\textbf{Soddisfacibile ma non valida}. Falsificabile con \(D=\{1,2\}\), \(B=\{1\}\), \(P=\{(1,1), (2,2)\}\).

\subsubsection*{(c) \((a=b) \to ((c=b) \to ((\forall x \st B(a,x)) \to (\exists x \st B(c,x))))\)}
\paragraph{Svolgimento:}
\textbf{Valida}. Assumendo dimostrati \(Symm\) e \(Trans\):
\begin{prooftree}
    \small
    \AxiomC{\({[\forall x \st B(a,x)]}_3\)}
    \RuleLabel{\(\forall E\)}
    \UnaryInfC{\(B(a,a)\)}
    \AxiomC{\({[a=b]}_1\)}
    \AxiomC{\({[c=b]}_2\)}
    \RuleLabel{\(Symm\)}
    \UnaryInfC{\(b=c\)}
    \RuleLabel{\(Trans\)}
    \BinaryInfC{\(a=c\)}
    \RuleLabel{\(Sub_\phi\)}
    \BinaryInfC{\(B(c,a)\)}
    \RuleLabel{\(\exists I\)}
    \UnaryInfC{\(\exists x \st B(c,x)\)}
    \RuleLabel{\(\to I_3\)}
    \UnaryInfC{\((\forall x \st B(a,x)) \to (\exists x \st B(c,x))\)}
    \RuleLabel{\(\to I_2\)}
    \UnaryInfC{\((c=b) \to ((\forall x \st B(a,x)) \to (\exists x \st B(c,x)))\)}
    \RuleLabel{\(\to I_1\)}
    \UnaryInfC{\((a=b) \to ((c=b) \to ((\forall x \st B(a,x)) \to (\exists x \st B(c,x))))\)}
\end{prooftree}
